## 一、关键技术与问题7 - 学习数据采集与分析系统的设计与实现

【描述系统设计阶段考虑到的可能遇到到的关键技术与问题，包括未知解决方法的问题、未使用过的技术等所有可能影响系统开发的点】

本系统需要实现完整的学习行为追踪、数据分析和可视化功能，为学生提供个性化的学习反馈和改进建议。核心挑战包括：如何在不影响主业务流程的前提下采集学习数据、如何设计高效的统计算法、如何实现前后端数据流的无缝对接、以及如何构建直观的数据可视化界面。

### 1.1 问题描述

1. **异步数据采集设计**：学习行为采集不能阻塞用户的正常操作（提问、代码提交），需要异步非阻塞设计
2. **数据模型设计复杂度**：需要同时追踪多种学习行为（提问、代码提交、知识查看、错误模式），数据库表结构设计需平衡范式与查询效率
3. **错误模式识别准确性**：如何从代码分析结果中准确提取错误类型、描述、相关知识点，并实现智能去重
4. **统计算法效率**：大量历史数据的聚合查询（按天统计、知识点分组）可能导致性能问题
5. **薄弱环节分析算法**：如何综合考虑正确率、频率、时间衰减等因素生成准确的薄弱点判定
6. **前端状态管理复杂度**：多个图表组件需要共享同一套数据，如何避免重复请求和状态不一致
7. **响应式布局适配**：左侧卡片列表与右侧图表区域的布局需要适配不同屏幕尺寸，同时保持交互流畅

### 1.2 可能的解决方案

**数据采集架构方案：**

|------|------|------|---------|

| 异步队列（Celery） | 高性能，可扩展 | 架构复杂，需要Redis/RabbitMQ | 高并发生产环境 |

```python
# backend/services/analytics/data_collector.py
class DataCollector:
    """学习数据采集服务"""

    @staticmethod
    def record_question(user_id, session_id, question, knowledge_topics=None):
        """记录用户提问行为"""
        try:
            record = LearningRecord(
                user_id=user_id,
                session_id=session_id,
                action_type='ask',
                content=json.dumps({
                    'question': question[:500]  # 限制长度
                }, ensure_ascii=False),
                knowledge_topics=','.join(knowledge_topics) if knowledge_topics else None
            )

            db.session.add(record)
            db.session.commit()

            print(f"✅ 记录提问行为: User {user_id}, Session {session_id}")
            return record

        except Exception as e:
            db.session.rollback()
            print(f"❌ 记录提问失败: {e}")
            return None

    @staticmethod
    def record_code_submission(user_id, session_id, code, result, knowledge_topics=None):
        """记录代码提交行为"""
        try:
            # 提取知识点（从代码特征）
            if not knowledge_topics:
                features = result.get('features', {})
                knowledge_topics = features.get('functions', [])

            # 判断正确性
            is_correct = result.get('score', 0) >= 60

            # 创建记录
            record = LearningRecord(
                user_id=user_id,
                session_id=session_id,
                action_type='code_submit',
                content=json.dumps({
                    'code': code[:1000],
                    'score': result.get('score'),
                    'level': result.get('level')
                }, ensure_ascii=False),
                knowledge_topics=','.join(knowledge_topics) if knowledge_topics else None,
                is_correct=is_correct
            )

            db.session.add(record)
            db.session.commit()

            # ⭐⭐⭐ 提取错误模式 ⭐⭐⭐
            DataCollector._extract_error_patterns(user_id, result)

            print(f"✅ 记录代码提交: User {user_id}, Score {result.get('score')}")
            return record

        except Exception as e:
            db.session.rollback()
            print(f"❌ 记录代码提交失败: {e}")
            return None

    @staticmethod
    def _extract_error_patterns(user_id, result):
        """从分析结果中提取错误模式"""
        try:
            # 从语法检查中提取错误
            syntax_check = result.get('syntax_check', {})
            if not syntax_check.get('valid'):
                errors = syntax_check.get('errors', [])
                for error in errors:
                    ErrorPattern.record_error(
                        user_id=user_id,
                        error_type='syntax',
                        error_description=error.get('message', ''),
                        related_topic=error.get('related_topic')
                    )

            # 从AI分析中提取问题
            ai_analysis = result.get('ai_analysis', {})
            problems = ai_analysis.get('problems', [])

            for problem in problems:
                severity = problem.get('severity', 'warning')
                if severity in ['error', 'warning']:
                    error_type = 'logic' if 'logic' in problem.get('description', '').lower() else 'concept'

                    ErrorPattern.record_error(
                        user_id=user_id,
                        error_type=error_type,
                        error_description=problem.get('description', ''),
                        related_topic=problem.get('related_topic')
                    )

        except Exception as e:
            print(f"⚠️ 提取错误模式失败: {e}")
```

#### 步骤1：实现核心统计算法

```python
# backend/services/analytics/stats_calculator.py
class StatsCalculator:
    """学习统计计算服务"""

    @staticmethod
    def get_user_overview(user_id, days=30):
        """获取用户学习概览"""
        cutoff_date = datetime.utcnow() - timedelta(days=days)

        # 1. 总学习时长（基于会话）
        sessions = ChatSession.query.filter(
            ChatSession.user_id == user_id,
            ChatSession.created_at >= cutoff_date
        ).all()

        total_duration = sum(
            (s.updated_at - s.created_at).total_seconds() / 3600
            for s in sessions if s.updated_at and s.created_at
        )

        # 2. 提问次数
        question_count = LearningRecord.query.filter_by(
            user_id=user_id,
            action_type='ask'
        ).filter(
            LearningRecord.created_at >= cutoff_date
        ).count()

        # 3. 代码提交次数和正确率
        code_records = LearningRecord.query.filter_by(
            user_id=user_id,
            action_type='code_submit'
        ).filter(
            LearningRecord.created_at >= cutoff_date
        ).all()

        code_count = len(code_records)
        correct_count = sum(1 for r in code_records if r.is_correct)
        accuracy = (correct_count / code_count * 100) if code_count > 0 else 0

        # 4. 活跃天数（使用 SQL DISTINCT）
        active_dates = db.session.query(
            func.date(LearningRecord.created_at)
        ).filter_by(
            user_id=user_id
        ).filter(
            LearningRecord.created_at >= cutoff_date
        ).distinct().all()

        active_days = len(active_dates)

        return {
            'total_duration_hours': round(total_duration, 1),
            'question_count': question_count,
            'code_count': code_count,
            'accuracy': round(accuracy, 1),
            'active_days': active_days,
            'avg_daily_questions': round(question_count / max(active_days, 1), 1)
        }

    @staticmethod
    def get_learning_trend(user_id, days=30):
        """获取学习趋势（按天统计）"""
        cutoff_date = datetime.utcnow() - timedelta(days=days)

        # SQL分组查询
        daily_stats = db.session.query(
            func.date(LearningRecord.created_at).label('date'),
            func.count(LearningRecord.id).label('count'),
            func.sum(
                case(
                    (LearningRecord.action_type == 'code_submit', 1),
                    else_=0
                )
            ).label('code_count'),
            func.sum(
                case(
                    (LearningRecord.is_correct == True, 1),
                    else_=0
                )
            ).label('correct_count')
        ).filter_by(
            user_id=user_id
        ).filter(
            LearningRecord.created_at >= cutoff_date
        ).group_by(
            func.date(LearningRecord.created_at)
        ).all()

        # 填充缺失日期
        result = []
        for i in range(days):
            date = (datetime.utcnow() - timedelta(days=days-i-1)).date()
            stat = next((s for s in daily_stats if s.date == date), None)

            if stat:
                accuracy = (stat.correct_count / stat.code_count * 100) if stat.code_count > 0 else 0
                result.append({
                    'date': date.isoformat(),
                    'total_count': stat.count,
                    'code_count': stat.code_count,
                    'correct_count': stat.correct_count,
                    'accuracy': round(accuracy, 1)
                })
            else:
                result.append({
                    'date': date.isoformat(),
                    'total_count': 0,
                    'code_count': 0,
                    'correct_count': 0,
                    'accuracy': 0
                })

        return result

    @staticmethod
    def get_knowledge_mastery(user_id, days=30):
        """获取知识点掌握情况"""
        cutoff_date = datetime.utcnow() - timedelta(days=days)

        records = LearningRecord.query.filter_by(
            user_id=user_id
        ).filter(
            LearningRecord.created_at >= cutoff_date,
            LearningRecord.knowledge_topics.isnot(None)
        ).all()

        # 统计每个知识点
        topic_stats = {}

        for record in records:
            topics = record.knowledge_topics.split(',') if record.knowledge_topics else []

            for topic in topics:
                topic = topic.strip()
                if not topic:
                    continue

                if topic not in topic_stats:
                    topic_stats[topic] = {
                        'topic': topic,
                        'total_count': 0,
                        'code_count': 0,
                        'correct_count': 0,
                        'view_count': 0
                    }

                topic_stats[topic]['total_count'] += 1

                if record.action_type == 'code_submit':
                    topic_stats[topic]['code_count'] += 1
                    if record.is_correct:
                        topic_stats[topic]['correct_count'] += 1
                elif record.action_type == 'view_knowledge':
                    topic_stats[topic]['view_count'] += 1

        # 计算掌握度
        result = []
        for topic, stats in topic_stats.items():
            mastery = (stats['correct_count'] / stats['code_count'] * 100) if stats['code_count'] > 0 else 0

            result.append({
                'topic': topic,
```

### 4.1 实验目的

验证 WeaknessAnalyzer 能否准确识别用户的薄弱知识点、高频错误和低表现区域，并生成合理的改进建议。

### 4.2 实验步骤

#### 步骤1：实现薄弱环节分析算法

```python
# backend/services/analytics/weakness_analyzer.py
class WeaknessAnalyzer:
    """薄弱环节分析器"""

    @staticmethod
    def analyze_weaknesses(user_id, days=30):
        """综合分析用户的薄弱环节"""
        cutoff_date = datetime.utcnow() - timedelta(days=days)

        analysis = {
            'weak_topics': WeaknessAnalyzer._analyze_weak_topics(user_id, cutoff_date),
            'frequent_errors': WeaknessAnalyzer._analyze_frequent_errors(user_id),
            'low_performance_areas': WeaknessAnalyzer._analyze_low_performance(user_id, cutoff_date),
            'suggestions': []
        }

        # 生成个性化建议
        analysis['suggestions'] = WeaknessAnalyzer._generate_suggestions(analysis)

        return analysis

    @staticmethod
    def _analyze_weak_topics(user_id, cutoff_date):
        """分析薄弱知识点（正确率 < 60%）"""
        records = LearningRecord.query.filter_by(
            user_id=user_id,
            action_type='code_submit'
        ).filter(
            LearningRecord.created_at >= cutoff_date,
            LearningRecord.knowledge_topics.isnot(None)
        ).all()

        topic_performance = {}

        for record in records:
            topics = record.knowledge_topics.split(',') if record.knowledge_topics else []

            for topic in topics:
                topic = topic.strip()
                if not topic:
                    continue

                if topic not in topic_performance:
                    topic_performance[topic] = {'total': 0, 'correct': 0, 'incorrect': 0}

                topic_performance[topic]['total'] += 1
                if record.is_correct:
                    topic_performance[topic]['correct'] += 1
                else:
                    topic_performance[topic]['incorrect'] += 1

        # 筛选薄弱点
        weak_topics = []

        for topic, stats in topic_performance.items():
            if stats['total'] < 3:  # 至少3次记录
                continue

            accuracy = (stats['correct'] / stats['total']) * 100

            if accuracy < 60:
                weak_topics.append({
                    'topic': topic,
                    'accuracy': round(accuracy, 1),
                    'total_attempts': stats['total'],
                    'correct_count': stats['correct'],
                    'severity': 'high' if accuracy < 40 else 'medium'
                })

        weak_topics.sort(key=lambda x: x['accuracy'])
        return weak_topics

    @staticmethod
    def _analyze_frequent_errors(user_id):
        """分析高频错误（出现 ≥ 3次）"""
        patterns = ErrorPattern.query.filter(
            ErrorPattern.user_id == user_id,
            ErrorPattern.occurrence_count >= 3
        ).order_by(
            ErrorPattern.occurrence_count.desc()
        ).limit(10).all()

        frequent_errors = []

        for pattern in patterns:
            days_since_last = None
            if pattern.last_seen:
                days_since_last = (datetime.utcnow() - pattern.last_seen).days

            frequent_errors.append({
                'error_type': pattern.error_type,
                'description': pattern.error_description,
                'occurrence_count': pattern.occurrence_count,
                'related_topic': pattern.related_topic,
                'days_since_last': days_since_last,
                'is_recent': days_since_last is not None and days_since_last <= 7
            })

        return frequent_errors

    @staticmethod
    def _generate_suggestions(analysis):
        """生成个性化改进建议"""
        suggestions = []

        # 建议1：针对薄弱知识点
        weak_topics = analysis.get('weak_topics', [])
        if weak_topics:
            top_weak = weak_topics[0]
            suggestions.append({
                'type': 'weak_topic',
                'priority': 'high',
                'title': f'重点复习：{top_weak["topic"]}',
                'description': f'该知识点正确率仅 {top_weak["accuracy"]}%，建议重点复习',
                'action': '查看知识库相关章节'
            })

        # 建议2：针对高频错误
        frequent_errors = analysis.get('frequent_errors', [])
        if frequent_errors:
            top_error = frequent_errors[0]

            priority = 'high' if top_error.get('is_recent') else 'medium'

            suggestions.append({
                'type': 'frequent_error',
                'priority': priority,
                'title': f'注意：{top_error["error_type"]}类错误',
                'description': f'"{top_error["description"]}" 已出现 {top_error["occurrence_count"]} 次',
                'action': '练习相关题目以加强理解'
            })

        # 建议3：鼓励
        if not suggestions:
            suggestions.append({
                'type': 'encouragement',
                'priority': 'low',
                'title': '保持良好状态',
                'description': '近期学习状态良好，继续保持！',
                'action': '可以尝试更有挑战性的题目'
            })

        return suggestions
```

### 7.3 性能优化经验

**已实现的优化：**

1. ✅ 数据库索引：user_id + action_type + created_at 复合索引

2. **统计算法模板**（时间序列分析）

   ```python
   # 1. SQL 分组查询
   daily_stats = db.query(...).group_by(date).all()

   # 2. 日期填充（补全缺失）
   for i in range(days):
       date = target_date - timedelta(i)
       stat = find(daily_stats, date) or default_value
       result.append(stat)

   # 3. 返回排序结果
   return result
   ```

   ```javascript
   // Store 定义

         isLoading.value = false;
       }
     }

   });
   ```
