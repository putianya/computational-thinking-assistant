```python
from sentence_transformers import SentenceTransformer

class VectorService:
    def __init__(self):
        self.model = SentenceTransformer('paraphrase-multilingual-MiniLM-L12-v2')
        self.client = chromadb.PersistentClient(path=CHROMADB_PERSIST_DIRECTORY)
        self.collection = self.client.get_or_create_collection("computational_thinking")

    def add_documents(self, documents, metadatas, ids):
        embeddings = self.model.encode(documents).tolist()
        # ChromaDB要求metadata值不能为None
        cleaned_metadatas = []
        for meta in metadatas:
            cleaned = {k: str(v) if v is not None else '' for k, v in meta.items()}
            cleaned_metadatas.append(cleaned)
        self.collection.add(
            documents=documents,
            embeddings=embeddings,
            metadatas=cleaned_metadatas,
            ids=ids
        )

    def search(self, query, top_k=5, threshold=0.5):
        query_embedding = self.model.encode([query]).tolist()
        results = self.collection.query(
            query_embeddings=query_embedding,
            n_results=top_k
        )
        return results
```

1. **设计分块算法**

```python
import re

def split_by_headers(content: str, source: str) -> list:
    """按Markdown标题分割文档"""
    chunks = []

    # 匹配二级到四级标题
    header_pattern = re.compile(r'^(#{2,4})\s+(.+?)[\s]*$', re.MULTILINE)
    headers = list(header_pattern.finditer(content))

    if not headers:
        # 无标题，整文档作为一个块
        return [{'content': content.strip(), 'source': source,
                 'chapter': '', 'section': '', 'level': 0}]

    for i, match in enumerate(headers):
        level = len(match.group(1))  # 标题级别
        title = match.group(2).strip()
        start = match.end()
        end = headers[i + 1].start() if i + 1 < len(headers) else len(content)

        chunk_content = content[start:end].strip()
        if chunk_content:
            chunks.append({
                'content': f"## {title}\n\n{chunk_content}",
                'source': source,
                'chapter': title if level == 2 else '',
                'section': title if level > 2 else '',
                'level': level
            })

    return chunks
```

**遇到的问题及解决：**

   - 问题：初始正则 `r'^(#{2,4})\s*(.+?)[\s]*$'` 无法匹配某些标题

   - 解决：添加预处理统一换行符，调整正则为 `r'^(#{2,4})\s+(.+?)[\s]*$'`

2. **一级标题被跳过**
   - 问题：设计时只匹配二级及以下标题
   - 原因：一级标题通常是文档标题，不应作为内容块
   - 结论：这是预期行为，保持不变

## 二、关键技术与问题2 - 智能问答路由分类系统

【描述系统设计阶段考虑到的可能遇到到的关键技术与问题，包括未知解决方法的问题、未使用过的技术等所有可能影响系统开发的点】

本系统需要区分用户问题类型，将编程/计算思维相关问题路由到RAG模式（使用知识库增强回答），将普通问题路由到普通对话模式。

### 2.1 问题描述

1. **问题分类准确性**：如何准确判断问题是否与编程/计算思维相关
2. **关键词覆盖度**：编程领域词汇繁多，如何确保关键词列表完整
3. **中文分词问题**：中文没有天然的词边界，"C语言"等词汇可能无法正确匹配
4. **模式选择效率**：分类逻辑需要高效，不能影响用户体验

### 2.2 可能的解决方案

| 方案           | 优点               | 缺点                     |
| -------------- | ------------------ | ------------------------ |
| 关键词匹配     | 简单快速、可解释   | 覆盖度有限、难以处理变体 |
| 正则表达式模式 | 灵活、支持模式匹配 | 复杂模式维护困难         |
| 机器学习分类器 | 泛化能力强         | 需要训练数据、实现复杂   |
| LLM意图识别    | 理解能力强         | 延迟高、成本高           |

```python
# backend/utils/question_classifier.py

# 编程相关关键词列表
PROGRAMMING_KEYWORDS = [
    # 语言
    'python', 'java', 'javascript', 'c++', 'c#', 'go', 'rust', 'php',
    'c语言', 'C语言',
    # 概念
    '算法', '数据结构', '函数', '变量', '循环', '条件', '数组', '链表',
    '指针', '递归', '排序', '二叉树', '堆栈', '队列',
    # 计算思维
    '计算思维', '编程', '代码', '程序', '调试', 'debug',
    '编译', '运行', '报错', '语法错误',
]

class QuestionClassifier:
    def classify(self, question: str) -> dict:
        question_lower = question.lower()

        # 检查是否包含编程关键词
        if self._contains_programming_topic(question_lower):
            return {
                'type': 'rag',
                'confidence': 0.8,
                'reason': '包含编程相关关键词'
            }

        return {
            'type': 'normal',
            'confidence': 0.6,
            'reason': '普通问题'
        }

    def _contains_programming_topic(self, question: str) -> bool:
        for keyword in PROGRAMMING_KEYWORDS:
            if keyword.lower() in question:
                return True
        return False
```

```python
def _contains_programming_topic(self, question: str) -> bool:

    for keyword in PROGRAMMING_KEYWORDS:
        if keyword.lower() in question.lower():
            return True
    # 模式匹配
    for pattern in PROGRAMMING_PATTERNS:
        if re.search(pattern, question):
            return True
    return False
```

```python
import re

PROGRAMMING_KEYWORDS = [
```
